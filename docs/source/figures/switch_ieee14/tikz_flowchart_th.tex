% TikZ GFL/GFM switching-strategy flowchart (Thai), following the implemented
% two-term severity index (ibr.SwitchableIbr6, index_mode='agsi_pp',
% w_V=w_f=0.5 on J_V/J_f) with the healthy per-bus V_ref, and the declared
% CASE_DEFINED chronology.
% Drawn in LaTeX so lettering matches the report body font (TH Sarabun PSK).
\begin{tikzpicture}[
  font=\small,
  node distance=7mm and 8mm,
  startstop/.style={rounded corners=8pt,draw,thick,align=center,
    fill=blue!8,minimum height=8mm,text width=34mm,font=\bfseries},
  proc/.style={draw,thick,align=center,fill=blue!8,
    minimum height=9mm,text width=40mm},
  dec/.style={diamond,draw,thick,align=center,fill=yellow!25,
    inner sep=1.5pt,aspect=2.1,text width=30mm},
  act/.style={draw,thick,align=center,fill=green!15,
    minimum height=8mm,text width=40mm},
  tx/.style={draw,thick,align=center,fill=red!12,
    minimum height=8mm,text width=40mm},
  conn/.style={circle,draw=red,thick,text=red,fill=white,inner sep=1.5pt,
    font=\bfseries},
  lab/.style={font=\footnotesize},
  arr/.style={-{Latex[length=2.6mm]},thick}]

% ---- left column: setup ----
\node[startstop] (start) {เริ่มต้น};
\node[proc,below=of start] (setup) {%
  กำหนดจุดทำงานและพารามิเตอร์:\\
  slack $1.00\angle0^{\circ}$ pu, แบนด์ $\pm5\%$\\
  ลง Q-limits (SOURCE\_DEFINED)\\
  $\Gamma_{on}=0.65$, $\Gamma_{off}=0.35$\\
  $T_{d,on}=0.10$ s, $T_{d,off}=1.00$ s\\
  น้ำหนัก severity $w_V=w_f=0.5$ ($J_V$,$J_f$)\\
  $V_{ref}$ = healthy per-bus PF, $f_0=60$ Hz};
\node[proc,below=of setup] (events) {%
  เหตุการณ์ CASE\_DEFINED:\\
  SG trip 20 s, load step 50 s\\
  fault 85--85.15 s, line trip 110 s\\
  restore 145 s, $T_{end}$ ตาม contract};
\node[conn,below=of events] (p1l) {P1};

\draw[arr] (start) -- (setup);
\draw[arr] (setup) -- (events);
\draw[arr] (events) -- (p1l);

% ---- right column: online loop (anchored to the right of the setup column) ----
\node[conn,right=34mm of start] (p1r) {P1};
\node[proc,below=of p1r] (sense) {%
  ทุกขั้นเวลา $t$ ($dt=0.10$ s): วัดที่ PCC\\
  $V, f$ (การตรวจความยอมรับได้อื่นแยกต่างหาก)};
\node[proc,below=of sense] (norm) {%
  normalize สองพจน์:\\
  $J_V, J_f$\\
  (base $0.10$ pu, $0.50$ Hz)};
\node[proc,below=of norm] (sev) {%
  $S=\operatorname{sat}_{[0,1]}\!\left(\tfrac12J_V+\tfrac12J_f\right)$};
\node[dec,below=of sev] (qev) {เหตุการณ์\\ CASE\_DEFINED?};
\node[dec,below=of qev] (qgfl) {โหมด GFL และ\\ $S\ge\Gamma_{on}$ ครบ $T_{d,on}$?};
\node[dec,below=of qgfl] (qgfm) {โหมด GFM และ\\ $S<\Gamma_{off}$ ครบ $T_{d,off}$?};
\node[dec,below=of qgfm] (qend) {$t\ge T_{end}$?};
\node[startstop,below=of qend] (stop) {สิ้นสุด};

% yes-branch action boxes to the right of the diamonds
\node[tx,right=10mm of qev] (txn) {atomic transaction:\\ SG trip $\to$ all GFM;\\ SG reclose $\to$ SG ถือ reference\\ (โหมด IBR ยังไม่เปลี่ยน)};
\node[act,right=10mm of qgfl] (swgfl) {สลับ GFL $\rightarrow$ GFM};
\node[act,right=10mm of qgfm] (swgfm) {atomic รายอุปกรณ์\\ GFM $\rightarrow$ GFL;\\ reset dwell ของตัวที่เหลือ};

% spine arrows
\draw[arr] (p1r) -- (sense);
\draw[arr] (sense) -- (norm);
\draw[arr] (norm) -- (sev);
\draw[arr] (sev) -- (qev);
\draw[arr] (qev) -- node[lab,right]{ไม่ใช่} (qgfl);
\draw[arr] (qgfl) -- node[lab,right]{ไม่ใช่} (qgfm);
\draw[arr] (qgfm) -- node[lab,right]{ไม่ใช่} (qend);
\draw[arr] (qend) -- node[lab,right]{ใช่} (stop);
% yes branches
\draw[arr] (qev) -- node[lab,above]{ใช่} (txn);
\draw[arr] (qgfl) -- node[lab,above]{ใช่} (swgfl);
\draw[arr] (qgfm) -- node[lab,above]{ใช่} (swgfm);
% yes-action outputs: feed a shared vertical channel EAST of the action boxes,
% then one clean arrow west into qend (no line passes through a box).
\coordinate (ret) at ($(txn.east)+(0.8,0)$);
\draw (txn.east) -| (ret |- qend.east);
\draw (swgfl.east) -| (ret |- qend.east);
\draw (swgfm.east) -| (ret |- qend.east);
\draw[arr] (ret |- qend.east) -- (qend.east);
% loop-back from qend (no) up its own OUTER channel to sense.
\coordinate (loopx) at ($(txn.east)+(1.7,0)$);
\draw[arr] (qend.east) -- (qend.east -| loopx)
  node[lab,pos=0.3,above]{ไม่ใช่\; $t+dt$} |- (sense.east);
\end{tikzpicture}
